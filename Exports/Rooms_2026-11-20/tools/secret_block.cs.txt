System.Func<string, Transform> FindNamed = n => { foreach (var t in UnityEngine.Object.FindObjectsByType<Transform>(FindObjectsInactive.Include, FindObjectsSortMode.None)) if (t.name == n && t.gameObject.scene.IsValid()) return t; return null; };
System.Action<Transform, float> SetHealth = (t, v) => { if (t == null) return; foreach (var f in t.GetComponents<PlayMakerFSM>()) { var h = f.FsmVariables.FindFsmFloat("health"); if (f.FsmName.EndsWith("Health") && h != null) h.Value = v; } };
System.Func<Vector3, float, int> CandiesNear = (p, r) => { int c = 0; foreach (var rb in UnityEngine.Object.FindObjectsByType<Rigidbody>(FindObjectsSortMode.None)) if (rb.name.StartsWith("Candy_") && Vector3.Distance(new Vector3(rb.position.x, 0f, rb.position.z), new Vector3(p.x, 0f, p.z)) < r) c++; return c; };
System.Func<string> Bools = () =>
{
    var sbB = new System.Text.StringBuilder();
    foreach (var f in Resources.FindObjectsOfTypeAll<PlayMakerFSM>())
    {
        if (!f.gameObject.scene.IsValid() || !(f.FsmName == "GAMESTAGES" || f.FsmName == "EnemyManager")) continue;
        foreach (var b in f.FsmVariables.BoolVariables) sbB.Append(f.gameObject.name + "." + b.Name + "=" + b.Value + ";");
    }
    return sbB.ToString();
};

System.Action<string, Vector3, Vector3, float, bool> ShotAt = (file, at, look, fov, noRoof) =>
{
    var camGo3 = new GameObject("__VIEW_CAM__");
    var cam3 = camGo3.AddComponent<Camera>();
    if (Camera.main != null) cam3.CopyFrom(Camera.main);
    camGo3.AddComponent<UnityEngine.Rendering.Universal.UniversalAdditionalCameraData>().renderPostProcessing = true;
    cam3.enabled = false; cam3.orthographic = false; cam3.fieldOfView = fov; cam3.nearClipPlane = 0.1f; cam3.farClipPlane = 120f;
    var off = new System.Collections.Generic.List<Renderer>();
    foreach (var r in origin.GetComponentsInChildren<Renderer>(false)) if (r.enabled) { r.enabled = false; off.Add(r); }
    bool fog = RenderSettings.fog;
    if (noRoof)
    {
        RenderSettings.fog = false;
        foreach (var k in UnityEngine.Object.FindObjectsByType<Game.Rooms.RoomKit>(FindObjectsSortMode.None))
            foreach (var r in k.GetComponentsInChildren<Renderer>(false))
            {
                bool roofPart = false; var q = r.transform;
                while (q != null && q != k.transform) { if (q.name == "Roof" || q.name.Contains("Roof_Sub")) roofPart = true; q = q.parent; }
                if (roofPart && r.enabled) { r.enabled = false; off.Add(r); }
            }
    }
    foreach (var mb in UnityEngine.Object.FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
        if (mb.GetType().Name == "LampFlicker") { var lamp = mb.GetType().GetField("lamp", flags).GetValue(mb) as Light; if (lamp != null) lamp.intensity = (float)mb.GetType().GetField("baseIntensity", flags).GetValue(mb); }
    camGo3.transform.position = at; camGo3.transform.LookAt(look);
    Shot(cam3, 1280, 720, file);
    foreach (var r in off) if (r != null) r.enabled = true;
    RenderSettings.fog = fog;
    UnityEngine.Object.Destroy(camGo3);
};

System.Action<string> RunSecret = gsState =>
{
    float t = Time.realtimeSinceStartup - stAt;
    if (st == 1)
    {
        foreach (var k in UnityEngine.Object.FindObjectsByType<Game.Rooms.SecretRoom>(FindObjectsSortMode.None)) secret = k;
        kitR = secret != null ? secret.GetComponent<Game.Rooms.RoomKit>() : null;
        Check("secret room on the bedroom (" + (secret != null) + ")", secret != null && kitR != null);
        var chained = FindNamed("Chained_Door");
        Check("chained door doorway centred at x -1.5 (" + (chained != null ? chained.parent.TransformPoint(new Vector3(-1.5f, 0f, 0f)).x.ToString("F2") : "none") + ")",
              chained != null && Mathf.Abs(chained.parent.TransformPoint(new Vector3(-1.5f, 0f, 0f)).x + 1.5f) < 0.05f);
        Check("before: fall edge covers the secret room's spot (0, 8)", Game.Locomotion.FallEdge.Covers(new Vector3(0f, 0.1f, 8f)));
        Check("W/C door leads onto the west fall edge (-4.8, -2.25)", Game.Locomotion.FallEdge.Covers(new Vector3(-4.8f, 0.1f, -2.25f)));
        Check("secret room shut before the door breaks", secret != null && !secret.IsOpen);
        SetHealth(chained, 0f);
        Log("chained door health -> 0 (broken)");
        st = 2; stAt = Time.realtimeSinceStartup;
    }
    else if (st == 2 && t > 4.5f)
    {
        Check("secret room opened", secret.IsOpen);
        Check("secret set up (walls stood)", kitR.IsUp("Secret"));
        Check("secret walls shown (" + SecretWallsShown() + " renderers)", SecretWallsShown() > 10);
        var fl = secret.transform.Find("Secret_Floor"); int flOn = 0; if (fl != null) foreach (var r in fl.GetComponentsInChildren<Renderer>()) if (r.enabled) flOn++;
        Check("secret floor shown (" + flOn + " tiles)", flOn == 9);
        var tp = secret.transform.Find("Secret Teleport Area");
        Check("secret teleport area on", tp != null && tp.gameObject.activeInHierarchy);
        Check("after: no fall edge in the secret room (0, 8)", !Game.Locomotion.FallEdge.Covers(new Vector3(0f, 0.1f, 8f)));
        Check("after: fall edge past its floor (0, 14)", Game.Locomotion.FallEdge.Covers(new Vector3(0f, 0.1f, 14f)));
        Check("after: fall edges beside it (+-5, 8)", Game.Locomotion.FallEdge.Covers(new Vector3(5f, 0.1f, 8f)) && Game.Locomotion.FallEdge.Covers(new Vector3(-5f, 0.1f, 8f)));
        var obstacle = typeof(Game.Rooms.SecretRoom).GetField("doorwayObstacle", flags).GetValue(secret) as UnityEngine.AI.NavMeshObstacle;
        Check("doorway obstacle off (" + (obstacle != null ? obstacle.transform.parent.name + "/" + obstacle.name : "none") + ")", obstacle != null && !obstacle.enabled);
        var p = secret.Pumpkin;
        float ph = 0f; if (p != null) { var rr = p.GetComponent<Renderer>(); if (rr != null) ph = rr.bounds.size.y; }
        Check("giant pumpkin standing, " + ph.ToString("F2") + " m tall", p != null && ph > 2.3f && ph < 2.95f);
        Check("light off before walking in", !secret.IsLit);
        var srObs = typeof(Game.Rooms.SecretRoom).GetField("doorwayObstacle", flags).GetValue(secret) as UnityEngine.AI.NavMeshObstacle;
        Log("doorway obstacle: SecretRoom's " + (srObs != null ? srObs.name + " #" + srObs.GetInstanceID() + " enabled " + srObs.enabled : "none") + "; found by the test " + (obstacle != null ? obstacle.name + " #" + obstacle.GetInstanceID() + " enabled " + obstacle.enabled : "none"));
        st = 3; stAt = Time.realtimeSinceStartup;
    }
    else if (st == 3 && t > 3.5f)
    {
        try { ShotAt("6_" + tagV + "_0dark_a_bedroom_to_secret", new Vector3(-0.9f, 1.65f, 0.6f), new Vector3(-1.5f, 1.25f, 7.5f), 60f, false); } catch (System.Exception pe) { Log("PHOTO ERROR " + pe.Message); }
        var obs2 = typeof(Game.Rooms.SecretRoom).GetField("doorwayObstacle", flags).GetValue(secret) as UnityEngine.AI.NavMeshObstacle;
        Check("doorway obstacle off (a few s later)", obs2 != null && !obs2.enabled);
        MovePlayerTo(new Vector3(-1.5f, body != null ? body.bounds.center.y : 1f, 6.2f));
        Log("player walked in through the doorway");
        st = 4; stAt = Time.realtimeSinceStartup;
    }
    else if (st == 4 && t > 1.5f)
    {
        Check("light on after walking in", secret.IsLit);
        try { Views("1lit"); } catch (System.Exception pe) { Log("PHOTO ERROR " + pe.Message); }
        var p = secret.Pumpkin;
        candyBase = CandiesNear(p != null ? p.transform.position : new Vector3(0f, 0f, 7.85f), 9f);
        boolsBefore = Bools();
        Game.Rooms.SecretRoom.ForceSpider = wantSpider;
        if (p != null)
        {
            var hd = p.GetComponent<Game.Combat.EnemyHitDamage>();
            bool hurt = hd != null && hd.TakeBlast(1f, p.transform.position + Vector3.up, 2f, true);
            Log("giant pumpkin: one blast of 1 as the player (" + hurt + ")");
        }
        st = 5; stAt = Time.realtimeSinceStartup;
    }
    else if (st == 5 && t > 1.2f)
    {
        var p = secret.Pumpkin;
        float h = -1f; if (p != null) foreach (var f in p.GetComponents<PlayMakerFSM>()) if (f.FsmName == "Health") h = f.FsmVariables.FindFsmFloat("health").Value;
        Check("giant pumpkin took 1 of its 2 health (health " + h + ")", p != null && Mathf.Approximately(h, 1f));
        if (p != null) { var hd = p.GetComponent<Game.Combat.EnemyHitDamage>(); if (hd != null) hd.TakeBlast(1f, p.transform.position + Vector3.up, 2f, true); }
        Log("giant pumpkin: second blast");
        st = 6; stAt = Time.realtimeSinceStartup;
    }
    else if (st == 6 && t > 3.5f)
    {
        Check("giant pumpkin broken, outcome '" + secret.Outcome + "'", secret.Outcome == (wantSpider ? "spider" : "candies") && secret.Pumpkin == null);
        if (!wantSpider)
        {
            int n = CandiesNear(new Vector3(0f, 0f, 7.85f), 9f) - candyBase;
            Check("a fountain of 33 candies (fountains seen: " + string.Join(",", fountains.ConvertAll(x => x.ToString()).ToArray()) + "; " + n + " more candies near)", fountains.Contains(33));
            try { Views("2after"); } catch (System.Exception pe) { Log("PHOTO ERROR " + pe.Message); }
            st = 8; stAt = Time.realtimeSinceStartup;
        }
        else
        {
            var sp = secret.Spider;
            Check("Giant Spider out (" + (sp != null ? sp.name : "none") + ")", sp != null);
            if (sp != null)
            {
                float hp = -1f; foreach (var f in sp.GetComponents<PlayMakerFSM>()) if (f.FsmName == "SpiderHealth") hp = f.FsmVariables.FindFsmFloat("health").Value;
                Check("Giant Spider health 33 (" + hp + ")", Mathf.Approximately(hp, 33f));
                Check("Giant Spider scale 2.0-2.4 (" + sp.transform.localScale.x.ToString("F2") + ")", sp.transform.localScale.x > 1.95f && sp.transform.localScale.x < 2.45f);
                var flagsF = typeof(Game.Enemies.SpiderBrain).GetField("setOnDeath", flags);
                var fl = flagsF != null ? flagsF.GetValue(sp.GetComponent<Game.Enemies.SpiderBrain>()) as System.Collections.ICollection : null;
                Check("Giant Spider sets no flag on death (" + (fl != null ? fl.Count : -1) + ")", fl != null && fl.Count == 0);
                try { Views("2spider"); } catch (System.Exception pe) { Log("PHOTO ERROR " + pe.Message); }
                candyBase = CandiesNear(sp.transform.position, 9f);
            }
            st = 7; stAt = Time.realtimeSinceStartup;
        }
    }
    else if (st == 7)
    {
        var sp = secret.Spider;
        var hd = sp != null ? sp.GetComponent<Game.Combat.EnemyHitDamage>() : null;
        if (hd != null && !hd.IsDead && t < 25f) { spiderAt = sp.transform.position; hd.TakeBlast(6f, sp.transform.position, 2f, true); blasts++; return; }
        if (stAt > 0f && t < 28f && (hd == null || hd.IsDead))
        {
            Log("Giant Spider down after " + blasts + " blasts of 6 (killed by player " + (hd != null && hd.KilledByPlayer) + ")");
            st = 71; stAt = Time.realtimeSinceStartup;
        }
        else if (t >= 28f) { Check("Giant Spider died", false); st = 8; stAt = Time.realtimeSinceStartup; }
    }
    else if (st == 71 && t > 3f)
    {
        int n = CandiesNear(spiderAt, 9f) - candyBase;
        Check("a fountain of 3 candies when the Giant Spider died (fountains seen: " + string.Join(",", fountains.ConvertAll(x => x.ToString()).ToArray()) + "; " + n + " more candies near, some from pumpkins it broke)", fountains.Contains(3));
        string after = Bools();
        Check("no stage/enemy-manager flag changed by its death", after == boolsBefore);
        if (after != boolsBefore) { var a = after.Split(';'); var b = boolsBefore.Split(';'); for (int i = 0; i < Mathf.Min(a.Length, b.Length); i++) if (a[i] != b[i]) Log("   changed: " + b[i] + " -> " + a[i]); }
        st = 8; stAt = Time.realtimeSinceStartup;
    }
    else if (st == 8 && t > 0.5f)
    {
        var wc = FindNamed("Locked_Door");
        SetHealth(wc, 0f);
        var hits = Physics.OverlapBox(new Vector3(-3.2f, 1f, -2.25f), new Vector3(0.45f, 0.9f, 0.7f), Quaternion.identity, ~0, QueryTriggerInteraction.Ignore);
        var names = new System.Collections.Generic.List<string>();
        foreach (var h in hits) { var n = Base(h.attachedRigidbody != null ? h.attachedRigidbody.name : h.name); if (!names.Contains(n)) names.Add(n); }
        Log("W/C door broken; in front of it inside the bedroom: " + string.Join(", ", names.ToArray()));
        Check("past the W/C doorway is the fall edge (-4.6 .. -6, z -2.25)", Game.Locomotion.FallEdge.Covers(new Vector3(-4.6f, 0.1f, -2.25f)) && Game.Locomotion.FallEdge.Covers(new Vector3(-6f, 0.1f, -2.25f)));
        st = 81; stAt = Time.realtimeSinceStartup;
    }
    else if (st == 81 && t > 2f)
    {
        try { ShotAt("6_" + tagV + "_3_wc_door_broken", new Vector3(0.5f, 1.7f, -1.0f), new Vector3(-3.8f, 1.1f, -2.25f), 60f, false); } catch (System.Exception pe) { Log("PHOTO ERROR " + pe.Message); }
        st = 10; stAt = Time.realtimeSinceStartup;
        Log("secret test: the bot plays on (breaks the stage pumpkin) to see the room fold");
    }
};
